wallpaperinho — Programmatic multi-monitor wallpaper automation.

USAGE:
  bin/wallpaperinho [options]

OPTIONS:
  --debug                 Enable debug overlay on generated images (overrides config)
  --noindex               Skip re-indexing images (use existing database)
  --recreate              Remove and recreate the database from scratch
  --profile <name>        Use a named profile; always wins over random selection.
                          ("default" selects the implicit first profile in profiles.js)
  --directory <path>      Specify an image directory to index (can be repeated).
                          Replaces only the selected profile's imageDirectories --
                          strategies, upscaler model/args etc. still come from the profile.
  --strategy <name>       Override matching strategy (random|colorHSL|colorPalette|contrast|canny|entropy|gemini)
  --silent                Unattended mode: hide banner/info/debug output; warnings and
                          errors stay visible as plain "[WARN]"/"[ERR]"-prefixed lines without ANSI colors (sized for cron mail capture);
                          each such line also names the active profile + effective selection pool (picked subdir under exclusive modes) for easier debugging
  --random                Pick a random eligible profile before generating. This is already
                          the default when no --profile is given; kept for explicit crontab lines.
                          Profiles may opt out with "excludeFromRandom: true".
  --cron                  Unattended mode: implies --silent + --random
  --list-profiles         List available profiles, marking which join the random rotation vs those
                           excluded from it ("excludeFromRandom"), and exit
  --help                  Show this help message and exit

EXAMPLES:
  bin/wallpaperinho                                  # random eligible profile (rotation-aware)
  bin/wallpaperinho --debug
  bin/wallpaperinho --noindex
  bin/wallpaperinho --profile liminal                # pin one specific profile
  bin/wallpaperinho --profile default                # pin the implicit first profile
  bin/wallpaperinho --directory /path/to/images      # custom dirs, profile settings still apply
  bin/wallpaperinho --directory /a --directory /b --recreate
  bin/wallpaperinho --strategy canny
  bin/wallpaperinho --profile liminal --strategy colorHSL
  bin/wallpaperinho --cron                           # quiet unattended run (ideal for crontab)
